Q1. Create a form UI which will take two inputs between number 5 and 10 and return their addition as result. Print that result in the UI.
 Use try/catch/throw to handle below errors here .. 
If a number >10 - “Wrong Input : number is too high” 
If a number < 5 - “Wrong Input : number is too low”
If input is no number print error - “Wrong Input  : input can only be a number”


<!DOCTYPE html>
<html>
<head>
  <title>Addition Form</title>
  <style>
    body {
      text-align: center;
    }
    input, button {
      padding: 10px;
      margin: 10px;
    }
    #result {
      margin-top: 20px;
      font-weight: bold;
    }
  </style>
</head>
<body>
  <h1>Add Two Numbers Between 5 And 10</h1>
  <input type="text" id="num1" placeholder="Enter first number">
  <span>+</span>
  <input type="text" id="num2" placeholder="Enter second number">
  <button onclick="calculate()">Add</button>
  <div id="result"></div>
  <script>
    function calculate(){
      const resultDiv = document.getElementById("result");
      resultDiv.innerHTML = "";
      try {
        let n1 = document.getElementById("num1").value;
        let n2 = document.getElementById("num2").value;
        
        if (isNaN(n1) || isNaN(n2) || n1 === "" || n2 === "") {
          throw "Wrong Input : input can only be a number";
        }
        n1 = Number(n1);
        n2 = Number(n2);

        if (n1>10 || n2>10) {
          throw "Wrong Input : number is too high";
        }
        if (n1<5 || n2<5) {
          throw "Wrong Input : number is too low";
        }
        
        const sum = n1 + n2;
        resultDiv.innerHTML="Result: " + sum;
      } catch (err) {
        resultDiv.innerHTML = err;
        resultDiv.style.color = "red";
      }
    }
  </script>
</body>
</html>
